# Budgets and Alerts (Step 7) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ทำขั้นที่ 7 ของ `SPEC.md`: ผู้ใช้ตั้งงบรายเดือนต่อหมวดรายจ่ายในหน้าเว็บ LIFF (ตั้งครั้งเดียวใช้ทุกเดือนจนกว่าจะแก้) และบอทเตือนต่อท้ายคำตอบตอนบันทึกเมื่อรายการนั้นทำให้หมวดข้าม 80% หรือ 100% ของงบ

**Architecture:** ตาราง `budgets` มีอยู่แล้ว (`month` เป็นวันที่ 1 ของเดือน, unique `(user_id, category_id, month)`) แถวหนึ่งแปลว่า "งบของหมวดนี้ตั้งแต่เดือนนั้นเป็นต้นไป" งบของเดือน M คือแถวล่าสุดที่ `month <= M`; `amount` เป็น null แปลว่าไม่ตั้งงบตั้งแต่เดือนนั้น SQL function `budget_status(user, month)` คืนทุกหมวดรายจ่ายพร้อมงบที่มีผลและยอดใช้จริงของเดือน ใช้ทั้งหน้าเว็บ (`GET /api/budgets`) และการเตือนในบอท การเตือนคิดจากยอดหลังบันทึกเทียบกับยอดก่อนบันทึก (หักรายการของข้อความนี้ออก) จึงเตือนเฉพาะตอนข้ามเส้นโดยไม่ต้องเก็บสถานะ

**Tech Stack:** Node.js 22 (CommonJS), Express 5.2.1, @supabase/supabase-js 2.117.2, LIFF SDK v2 (CDN), Vitest 5.0.2

```sql
-- รันใน Supabase SQL Editor ต่อจาก 004 เพื่อเปิดใช้งบประมาณรายเดือนต่อหมวด

-- แถวหนึ่งคืองบตั้งแต่เดือนนั้นเป็นต้นไป amount เป็น null แปลว่ายกเลิกงบตั้งแต่เดือนนั้น (check เดิมผ่านเมื่อค่าเป็น null)
alter table public.budgets alter column amount drop not null;

create or replace function public.budget_status(p_user_id uuid, p_month date)
returns table (category_id uuid, category text, budget numeric, spent numeric)
language sql
stable
set search_path = public
as $$
  select c.id, c.name, b.amount, coalesce(s.spent, 0)
  from public.categories c
  left join lateral (
    select b.amount
    from public.budgets b
    where b.user_id = p_user_id
      and b.category_id = c.id
      and b.month <= p_month
    order by b.month desc
    limit 1
  ) b on true
  left join lateral (
    select sum(t.amount) as spent
    from public.transactions t
    where t.user_id = p_user_id
      and t.category_id = c.id
      and t.type = 'expense'
      and t.occurred_on >= p_month
      and t.occurred_on < (p_month + interval '1 month')::date
  ) s on true
  where c.user_id = p_user_id
    and c.type = 'expense'
  order by c.name
$$;

-- function ใหม่ใน Postgres เรียกได้ทุก role โดยค่าเริ่มต้น จึงปิดไว้อีกชั้นเหมือน summarize_transactions
revoke execute on function public.budget_status(uuid, date) from public, anon, authenticated;
grant execute on function public.budget_status(uuid, date) to service_role;
```

```js
  async function getBudgetStatus(userId, month) {
    const { data, error } = await supabase.rpc('budget_status', { p_user_id: userId, p_month: `${month}-01` });
    throwIfError('getBudgetStatus', error);
    return data.map((row) => ({
      categoryId: row.category_id,
      category: row.category,
      budget: row.budget === null ? null : Number(row.budget),
      spent: Number(row.spent),
    }));
  }

  // แถวของเดือนนี้ทับค่าเดิมของเดือนเดียวกัน ส่วนเดือนก่อนหน้ายังใช้งบเดิม
  async function setBudget({ userId, categoryId, month, amount }) {
    const { error } = await supabase
      .from('budgets')
      .upsert(
        { user_id: userId, category_id: categoryId, month: `${month}-01`, amount },
        { onConflict: 'user_id,category_id,month' }
      );
    throwIfError('setBudget', error);
  }
```

```js
const { formatAmount } = require('../parser/format-reply');

// เทียบเป็นสตางค์แบบจำนวนเต็ม กันทศนิยมของ float ทำให้ข้ามเส้นผิด
function toSatang(amount) {
  return Math.round(amount * 100);
}

function monthOf(isoDate) {
  return isoDate.slice(0, 7);
}

function budgetMonths(rows) {
  return [...new Set(rows.filter((row) => row.type === 'expense').map((row) => monthOf(row.occurred_on)))];
}

function findBudgetAlerts(rows, statusByMonth) {
  const added = new Map();
  for (const row of rows) {
    if (row.type !== 'expense') continue;
    const key = `${monthOf(row.occurred_on)}|${row.category_id}`;
    added.set(key, (added.get(key) || 0) + toSatang(row.amount));
  }
  const alerts = [];
  for (const [key, addedSatang] of added) {
    const [month, categoryId] = key.split('|');
    const status = (statusByMonth.get(month) || []).find((item) => item.categoryId === categoryId);
    if (!status || status.budget === null) continue;
    const budget = toSatang(status.budget);
    const after = toSatang(status.spent);
    // ยอดจาก SQL นับรายการที่เพิ่งบันทึกแล้ว ยอดก่อนหน้าจึงต้องหักออก
    const before = after - addedSatang;
    // เตือนเฉพาะข้อความที่ทำให้ข้ามเส้น รายการถัดไปจึงไม่เตือนซ้ำโดยไม่ต้องเก็บสถานะ
    let level = null;
    if (before < budget && after >= budget) {
      level = 'over';
    } else if (before * 5 < budget * 4 && after * 5 >= budget * 4) {
      level = 'warn';
    }
    if (level) {
      alerts.push({ level, month, category: status.category, spent: status.spent, budget: status.budget });
    }
  }
  return alerts;
}

function formatBudgetAlerts(alerts) {
  return alerts
    .map((alert) => {
      const [year, month] = alert.month.split('-');
      const percent = Math.floor((toSatang(alert.spent) * 100) / toSatang(alert.budget));
      const label = alert.level === 'over' ? 'เกินงบ' : 'ใกล้เต็มงบ';
      return `${label} ${alert.category} เดือน ${month}/${year}: ใช้ไป ${formatAmount(alert.spent)} จาก ${formatAmount(alert.budget)} บาท (${percent}%)`;
    })
    .join('\n');
}

module.exports = { budgetMonths, findBudgetAlerts, formatBudgetAlerts };
```

```js
export function formatMonthLabel(month) {
  const [year, value] = month.split('-');
  return `${value}/${year}`;
}

// ปัด % ลงและเทียบเป็นสตางค์ ให้ตรงกับเส้นที่บอทใช้เตือนในแชต
export function budgetRows(budgets) {
  const rows = budgets.map((row) => {
    if (row.budget === null) {
      return {
        categoryId: row.categoryId,
        category: row.category,
        budget: null,
        spent: row.spent,
        level: 'none',
        percent: null,
        width: 0,
        text: `ใช้ไป ${formatBaht(row.spent)} · ยังไม่ตั้งงบ`,
      };
    }
    const percent = Math.floor((Math.round(row.spent * 100) * 100) / Math.round(row.budget * 100));
    return {
      categoryId: row.categoryId,
      category: row.category,
      budget: row.budget,
      spent: row.spent,
      level: percent >= 100 ? 'over' : percent >= 80 ? 'warn' : 'ok',
      percent,
      width: Math.min(percent, 100),
      text: `ใช้ไป ${formatBaht(row.spent)} จาก ${formatBaht(row.budget)} (${percent}%)`,
    };
  });
  // หมวดที่ตั้งงบแล้วขึ้นก่อน เพราะเป็นส่วนที่ผู้ใช้ต้องติดตาม
  return [...rows.filter((row) => row.budget !== null), ...rows.filter((row) => row.budget === null)];
}

export function describeBudgetFailure(status) {
  if (status === 401) return LOGIN_REQUIRED_MESSAGE;
  if (status === 400) return 'จำนวนเงินไม่ถูกต้อง ใส่ได้ไม่เกิน 10,000,000 บาท ทศนิยมไม่เกิน 2 ตำแหน่ง';
  if (status === 404) return 'ไม่พบหมวดนี้แล้ว';
  return 'บันทึกงบไม่สำเร็จ ลองใหม่อีกครั้ง';
}
```

```js
function setBudgetBusy(isBusy) {
  budgetBusy = isBusy;
  els.budgetSave.disabled = isBusy;
  els.budgetRemove.disabled = isBusy;
  els.budgetCancel.disabled = isBusy;
  els.budgetError.textContent = isBusy ? 'กำลังบันทึก...' : '';
  els.budgetError.hidden = !isBusy;

async function saveBudget(amount) {
  if (budgetBusy) return;
  const target = editingBudget;
  setBudgetBusy(true);
  let failure = null;
  try {
    await api.setBudget(target.categoryId, { month: target.month, amount });

    failure = describeBudgetFailure(err instanceof ApiError ? err.status : undefined);

    setBudgetBusy(false);

    els.budgetError.hidden = false;
    return;
  }
  els.budgetEditor.close();
  await loadBudgets();
}
```

| 12 | (ขอบเดือน) บันทึกรายจ่ายหมวดอาหาร 1 รายการลงวันสุดท้ายของเดือนที่แล้ว และอีก 1 รายการลงวันแรกของเดือนหน้า: ในแชตใช้ `เมื่อวาน` ได้เฉพาะวันก่อนวันนี้ (parser แปลงวันที่ด้วย Claude และรองรับคำอย่าง `เมื่อวาน`) ถ้าต้องการวันอื่นให้บันทึกแล้วแก้วันที่ของรายการในหน้าเว็บด้วย dialog แก้ไข | แต่ละรายการนับใน `ใช้ไป` ของเดือนของมันเท่านั้น |
| 13 | (รายรับไม่นับ) บันทึกรายรับ 1 รายการ | `ใช้ไป` ของแถวงบรายจ่ายทุกแถวไม่เปลี่ยน |
| 14 | (ตั้งใหม่หลังหยุด) ในเดือนหนึ่งกด `ไม่ตั้งงบ` แล้วตั้งยอดใหม่ในเดือนเดียวกัน | ยอดใหม่ใช้งาน |
| 15 | (ตั้งทับเดือนเดียวกัน) ตั้งงบ `อาหาร` สองครั้งในเดือนเดียวกัน | แสดงเฉพาะยอดหลังสุด และตาราง `budgets` มีแถวของเดือนนั้นแถวเดียว |
| 16 | (ข้อความเช็กงบพัง ไม่บังคับ) ทดสอบไม่ได้ถ้าไม่ทำให้ฐานข้อมูลพัง | ครอบคลุมด้วย unit test ใน `src/bot.test.js` แล้ว |

## ข้อจำกัดที่ตั้งใจเลื่อนไปขั้นถัดไป

- ตั้งงบผ่านแชต (เช่นพิมพ์ `งบ อาหาร 5000`) ยังไม่ทำ ตั้งได้เฉพาะในหน้าเว็บ
- แก้หรือลบรายการในหน้าเว็บจนข้ามเส้นงบไม่มีการเตือนในแชต หน้าเว็บแสดงสถานะงบใหม่แทน
- ยกเลิกรายการด้วยปุ่ม `ยกเลิก` หลังเตือนแล้ว ถ้าบันทึกใหม่จนข้ามเส้นอีกจะเตือนอีกครั้ง
- ยังไม่มีงบรวมทั้งเดือน (ข้ามหมวด) และยังไม่แสดงงบในการ์ด `สรุป`
- ข้อความของหมวดเดียวกันใน webhook request เดียวที่ประมวลผลพร้อมกัน (`Promise.allSettled`) อาจเตือนซ้ำหรือข้ามเตือน (ข้อจำกัดที่ยอมรับ สาเหตุ: แต่ละข้อความหักเฉพาะยอดของตัวเองออกจากยอดรวมใน SQL)
- month นอกช่วง 2000-2100 หรือ 0000/9999 ที่ API รับไว้อาจทำให้ DB error เรื่องวันที่และตอบ 500 (เปิด ticket แล้ว)
- ข้อบกพร่องเล็กของ UI ใน Task 6 เลื่อนไปก่อน: `loadBudgets` ครั้งที่สองที่มาช้ากว่าเขียนทับ, focus หลังบันทึก, ข้อความ saving ใช้ element เดียวกับข้อความเตือน, ไม่มี retry เมื่อโหลดไม่สำเร็จ, layout ขยับตอนโหลดช้า, หัวข้อ dialog ตัดบรรทัด, กด Esc สองครั้งตอน busy
