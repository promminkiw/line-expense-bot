# กระเป๋าเงิน เฟส 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ผู้ใช้มีกระเป๋า "เงินสด" ตั้งแต่แรก สร้าง/เปลี่ยนชื่อ/ซ่อนกระเป๋าเพิ่มในแท็บโปรไฟล์ ดูยอดแยกกระเป๋า พิมพ์ "กาแฟ 60 บัตรเครดิต" แล้ว Claude เลือกกระเป๋าให้ แก้กระเป๋าของรายการและกรองรายการตามกระเป๋าใน LIFF และถามยอดแยกกระเป๋าได้

**Architecture:** ตาราง `wallets` ใหม่และคอลัมน์ `transactions.wallet_id` (null = กระเป๋าตั้งต้น "เงินสด" ไม่ backfill รายการเก่า) ใน `supabase/024_wallets.sql` พร้อม SQL function `wallet_totals` และ `query_totals` ที่รับ `p_wallet` เพิ่ม ค่าคงที่และ helper ของกระเป๋าอยู่ใน `src/wallets/wallets.js` ที่เดียว parser และ question parser ได้รายชื่อกระเป๋าเฉพาะผู้ใช้ที่มีมากกว่าหนึ่งกระเป๋า (ผู้ใช้กระเป๋าเดียว prompt และ schema เหมือนเดิมทุกตัวอักษร) LIFF มีส่วน "กระเป๋าเงิน" ในแท็บโปรไฟล์ และตัวเลือก/ตัวกรองกระเป๋าที่โผล่เฉพาะผู้ใช้หลายกระเป๋า

**Tech Stack:** Node.js 24 + Express 5, @line/bot-sdk, @anthropic-ai/sdk (claude-haiku-5-5), Supabase Postgres, LIFF (vanilla JS ES modules ใน `public/liff/`), vitest + PGlite + jsdom

### Task 1: SQL กระเป๋าเงิน (024)

**Depends on:** none

**Files:**
- Create: `supabase/024_wallets.sql`
- Modify: `supabase/schema.sql:1` (บรรทัดหัวไฟล์ที่บอกลำดับการรัน)
- Test: `src/db/sql-functions.test.js` (`MIGRATIONS`, describe ใหม่, แถวใน `function permissions`)

**Interfaces:**
- Produces:
  - ตาราง `public.wallets (id uuid, user_id uuid, name text 1-30 ตัวอักษร, hidden boolean not null default false, created_at)` unique `(user_id, name)` และ `(id, user_id)`; ผู้ใช้เดิมทุกคนมีแถว `name = 'เงินสด'`
  - `public.transactions.wallet_id uuid null` FK `(wallet_id, user_id)` -> `wallets (id, user_id)` ชื่อ constraint `transactions_wallet_fkey`; null = กระเป๋าตั้งต้น
  - `public.wallet_totals(p_user_id uuid)` returns table `(wallet_id uuid, name text, hidden boolean, income numeric, expense numeric, entry_count bigint)` หนึ่งแถวต่อกระเป๋าของผู้ใช้ (รวมกระเป๋าที่ยังไม่มีรายการ) รายการที่ `wallet_id` เป็น null นับเข้ากระเป๋าชื่อ `เงินสด` เรียงกระเป๋าตั้งต้นก่อน แล้วตาม created_at และชื่อ
  - `public.query_totals(p_user_id uuid, p_from date, p_to date, p_type text, p_category text, p_keyword text, p_wallet text default '')` returns table `(type text, total numeric, entry_count bigint)`; `p_wallet = ''` คือไม่กรอง ถ้าไม่ว่างเทียบชื่อกระเป๋าตรงตัว โดยรายการ null มีชื่อ `เงินสด`; มีแค่ overload เดียว

  }

    const first = await createUser('U1');
    const second = await createUser('U2');

-- รันใน Supabase SQL Editor ต่อจาก 023 ก่อน push โค้ดกระเป๋าเงิน รันซ้ำได้ และโค้ดเดิมที่ deploy อยู่ยังใช้ได้หลังรัน

create table if not exists public.wallets (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.users (id) on delete cascade,
  name text not null check (char_length(name) between 1 and 30),
  hidden boolean not null default false,
  created_at timestamptz not null default now(),
  unique (user_id, name),
  -- ให้ transactions อ้างอิง (id, user_id) คู่กันได้ กันการใช้กระเป๋าของผู้ใช้คนอื่น
  unique (id, user_id)
);

-- ไม่สร้าง policy เหมือนตารางอื่น: anon key เข้าถึงไม่ได้ ส่วน server ใช้ service role
alter table public.wallets enable row level security;

-- ผู้ใช้เดิมได้กระเป๋าตั้งต้นทันที คำถามยอดจะได้รู้จักชื่อ เงินสด โดยไม่ต้องรอให้โค้ดซ่อม
insert into public.wallets (user_id, name)
select id, 'เงินสด' from public.users
on conflict (user_id, name) do nothing;

alter table public.transactions add column if not exists wallet_id uuid;

alter table public.transactions drop constraint if exists transactions_wallet_fkey;
alter table public.transactions
  add constraint transactions_wallet_fkey foreign key (wallet_id, user_id) references public.wallets (id, user_id);

-- กระเป๋าที่ยังไม่มีรายการก็ต้องขึ้น จึงเริ่มจาก wallets แล้ว left join
create or replace function public.wallet_totals(p_user_id uuid)
returns table (wallet_id uuid, name text, hidden boolean, income numeric, expense numeric, entry_count bigint)
language sql
stable
set search_path = public
as $$
  select
    w.id,
    w.name,
    w.hidden,
    coalesce(sum(t.amount) filter (where t.type = 'income'), 0),
    coalesce(sum(t.amount) filter (where t.type = 'expense'), 0),
    count(t.id)
  from public.wallets w
  left join public.transactions t
    on t.user_id = w.user_id
    -- wallet_id null คือกระเป๋าตั้งต้น ชื่อต้องตรงกับ DEFAULT_WALLET_NAME ใน src/wallets/wallets.js
    and (t.wallet_id = w.id or (t.wallet_id is null and w.name = 'เงินสด'))
  where w.user_id = p_user_id
  group by w.id, w.name, w.hidden, w.created_at
  order by w.name <> 'เงินสด', w.created_at, w.name
$$;

-- create or replace ที่เพิ่มพารามิเตอร์จะได้ overload ใหม่ จึงลบตัว 6 พารามิเตอร์ก่อน
drop function if exists public.query_totals(uuid, date, date, text, text, text);

-- p_wallet มีค่า default ให้โค้ดเดิมที่เรียก 6 พารามิเตอร์ยังใช้ได้ระหว่างรัน SQL ก่อน push
create or replace function public.query_totals(
  p_user_id uuid,
  p_from date,
  p_to date,
  p_type text,
  p_category text,
  p_keyword text,
  p_wallet text default ''
)
returns table (type text, total numeric, entry_count bigint)
language sql
stable
set search_path = public
as $$
  select t.type, sum(t.amount), count(*)
  from public.transactions t
  join public.categories c on c.id = t.category_id and c.user_id = t.user_id
  left join public.wallets w on w.id = t.wallet_id and w.user_id = t.user_id
  where t.user_id = p_user_id
    and t.occurred_on between p_from and p_to
    and (p_type = 'both' or t.type = p_type)
    and (p_category = '' or c.name = p_category)
    -- คำค้นมาจากข้อความผู้ใช้ จึง escape \ % _ ให้เป็นตัวอักษรธรรมดา (ilike ใช้ \ เป็น escape โดยค่าเริ่มต้น)
    and (
      p_keyword = ''
      or t.note ilike '%' || replace(replace(replace(p_keyword, '\', '\\'), '%', '\%'), '_', '\_') || '%'
    )
    -- wallet_id null คือกระเป๋าตั้งต้น ชื่อต้องตรงกับ DEFAULT_WALLET_NAME ใน src/wallets/wallets.js
    and (p_wallet = '' or coalesce(w.name, 'เงินสด') = p_wallet)
  group by t.type
  order by t.type
$$;

-- function ใหม่ใน Postgres เรียกได้ทุก role โดยค่าเริ่มต้น จึงปิดไว้อีกชั้นเหมือน function อื่น
revoke execute on function public.wallet_totals(uuid) from public, anon, authenticated;
grant execute on function public.wallet_totals(uuid) to service_role;
revoke execute on function public.query_totals(uuid, date, date, text, text, text, text) from public, anon, authenticated;
grant execute on function public.query_totals(uuid, date, date, text, text, text, text) to service_role;

notify pgrst, 'reload schema';

function walletChoices(wallets) {
  const names = wallets.filter((wallet) => !wallet.hidden).map((wallet) => wallet.name);
  if (names.length < 2) return [];
  return [DEFAULT_WALLET_NAME, ...names.filter((name) => !isDefaultWallet(name))];
}

function walletQuestionNames(wallets) {

  return names.length < 2 ? [] : names;

function walletIdsByName(wallets) {
  return new Map(
    wallets.filter((wallet) => !wallet.hidden && !isDefaultWallet(wallet.name)).map((wallet) => [wallet.name, wallet.id])
  );
}

      .from('wallets')
      .select(WALLET_COLUMNS)

      .order('created_at')

  // ignoreDuplicates กันสมัครพร้อมกันหลาย event แล้วชน unique (user_id, name)
  async function seedDefaultWallet(userId) {
    const { error } = await supabase
      .from('wallets')
      .upsert({ user_id: userId, name: DEFAULT_WALLET_NAME }, { onConflict: 'user_id,name', ignoreDuplicates: true });
    throwIfError('seedDefaultWallet', error);
  }

  // ชื่อซ้ำชน unique (user_id, name) จึงคืน duplicate ให้ API ตอบ 409
  async function createWallet(userId, { name }) {
    const { data, error } = await supabase
      .from('wallets')
      .insert({ user_id: userId, name })
      .select(WALLET_COLUMNS)
      .single();
    if (error && error.code === UNIQUE_VIOLATION) {
      return { status: 'duplicate' };
    }
    throwIfError('createWallet', error);
    return { status: 'ok', wallet: toWallet(data) };
  }

  async function updateWallet(userId, id, fields) {
    const changes = {};
    if (fields.name !== undefined) changes.name = fields.name;
    if (fields.hidden !== undefined) changes.hidden = fields.hidden;
    const { data, error } = await supabase
      .from('wallets')
      .update(changes)
      .eq('user_id', userId)
      .eq('id', id)
      .select(WALLET_COLUMNS);
    if (error && error.code === UNIQUE_VIOLATION) {
      return { status: 'duplicate' };
    }
    throwIfError('updateWallet', error);
    return data.length > 0 ? { status: 'ok', wallet: toWallet(data[0]) } : { status: 'not_found' };
  }

  if (typeof value !== 'string') {

function validateWalletCreate(body) {
  if (!body || typeof body !== 'object') {
    return { ok: false, error: 'Invalid body' };
  }
  const name = validateWalletName(body.name);
  if (!name.ok) return name;
  return { ok: true, value: { name: name.value } };

function validateWalletUpdate(body) {
  if (!body || typeof body !== 'object') {
    return { ok: false, error: 'Invalid body' };
  }
  const value = {};
  if (body.name !== undefined) {
    const name = validateWalletName(body.name);
    if (!name.ok) return name;
    value.name = name.value;
  }
  if (body.hidden !== undefined) {
    if (typeof body.hidden !== 'boolean') {
      return { ok: false, error: 'Invalid hidden' };
    }
    value.hidden = body.hidden;
  }
  if (Object.keys(value).length === 0) {
    return { ok: false, error: 'Invalid body' };
  }
  return { ok: true, value };

export function describeWalletError(status, error) {
  if (status === 401) return LOGIN_REQUIRED_MESSAGE;
  if (status === 404) return 'ไม่พบกระเป๋านี้แล้ว ลองเปิดหน้าใหม่';

    canEdit: !wallet.isDefault,
    hideLabel: wallet.hidden ? 'เลิกซ่อน' : 'ซ่อน',

export function sortWallets(wallets) {
  return [...wallets.filter((wallet) => wallet.isDefault), ...wallets.filter((wallet) => !wallet.isDefault)];

    } finally {
      busy = false;
    }

export function filterTransactions(transactions, { query = '', categoryId = '', type = '', walletId = '' } = {}) {
  const needle = query.trim().toLowerCase();
  return transactions.filter((item) => {
    if (type && item.type !== type) return false;
    if (categoryId && item.categoryId !== categoryId) return false;
    if (walletId && item.walletId !== walletId) return false;
    if (!needle) return true;
    return `${item.categoryName} ${item.note || ''}`.toLowerCase().includes(needle);
  });
}

export function isFilterActive({ query = '', categoryId = '', type = '', walletId = '' } = {}) {
  return Boolean(query.trim() || categoryId || type || walletId);
}
